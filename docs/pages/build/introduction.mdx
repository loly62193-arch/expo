const GameScreen = ({ navigation }) => {
  const [balance, setBalance] = useState(0);
  const [betAmount, setBetAmount] = useState('10');
  const [isSpinning, setIsSpinning] = useState(false);
  const [slotResult, setSlotResult] = useState(['🍒', '🍒', '🍒']);
  const [lastWin, setLastWin] = useState(0);

  useEffect(() => {
    loadBalance();
  }, []);

  const loadBalance = async () => {
    const data = await AsyncStorage.getItem('user_data');
    if (data) setBalance(JSON.parse(data).coinBalance || 1000);
  };

  const updateBalance = async (newBalance) => {
    setBalance(newBalance);
    const data = await AsyncStorage.getItem('user_data');
    if (data) {
      const user = JSON.parse(data);
      user.coinBalance = newBalance;
      await AsyncStorage.setItem('user_data', JSON.stringify(user));
    }
  };

  const calculateResult = (bet) => {
    const symbols = ['🍒', '🍋', '🍊', '7️⃣', '💎', '🔔', '⭐'];
    const result = [
      symbols[Math.floor(Math.random() * symbols.length)],
      symbols[Math.floor(Math.random() * symbols.length)],
      symbols[Math.floor(Math.random() * symbols.length)]
    ];
    
    let multiplier = 0;
    if (result[0] === result[1] && result[1] === result[2]) {
      switch(result[0]) {
        case '7️⃣': multiplier = 50; break;
        case '💎': multiplier = 30; break;
        case '⭐': multiplier = 20; break;
        case '🔔': multiplier = 15; break;
        default: multiplier = 10;
      }
    } else if (result[0] === result[1] || result[1] === result[2] || result[0] === result[2]) {
      multiplier = 2;
    }
    
    return { result, win: bet * multiplier, multiplier };
  };

  const handleSpin = () => {
    const betValue = parseInt(betAmount) || 0;
    
    if (betValue < 1 || betValue > 10000) {
      Alert.alert('خطأ', 'مبلغ الرهان يجب أن يكون بين 1 و 10000');
      return;
    }
    
    if (balance < betValue) {
      Alert.alert('رصيد غير كافي', 'قم بإيداع كوين للمتابعة');
      return;
    }

    setIsSpinning(true);
    setLastWin(0);

    setTimeout(() => {
      const { result, win } = calculateResult(betValue);
      const newBalance = balance - betValue + win;
      
      setSlotResult(result);
      setLastWin(win);
      updateBalance(newBalance);
      setIsSpinning(false);
      
      if (win > 0) {
        Alert.alert('🎉 فوز!', `لقد ربحت ${win} كوين!`);
      }
    }, 1500);
  };

  const handleCollect = () => {
    Alert.alert('تم', 'تم إضافة الأرباح إلى رصيدك');
  };

  return (
    <View style={gameStyles.container}>
      <View style={gameStyles.balanceContainer}>
        <Text style={gameStyles.balanceText}>💰 {balance.toLocaleString()} كوين</Text>
      </View>

      <View style={gameStyles.gameContainer}>
        <SlotMachine result={slotResult} isSpinning={isSpinning} />
      </View>

      {lastWin > 0 && !isSpinning && (
        <View style={gameStyles.winContainer}>
          <Text style={gameStyles.winText}>+{lastWin} كوين!</Text>
        </View>
      )}

      <View style={gameStyles.betContainer}>
        <Text style={gameStyles.betLabel}>مبلغ الرهان:</Text>
        <View style={gameStyles.betInputContainer}>
          <TextInput
            style={gameStyles.betInput}
            value={betAmount}
            onChangeText={setBetAmount}
            keyboardType="numeric"
            maxLength={5}
            editable={!isSpinning}
          />
          <Text style={gameStyles.betUnit}>كوين</Text>
        </View>
        <Text style={gameStyles.betRange}>من 1 إلى 10000</Text>
      </View>

      <View style={gameStyles.buttonsContainer}>
        <TouchableOpacity 
          style={[gameStyles.button, gameStyles.collectButton]} 
          onPress={handleCollect}
          disabled={isSpinning}>
          <Text style={gameStyles.buttonText}>💰 اسحب</Text>
        </TouchableOpacity>

        <TouchableOpacity 
          style={[gameStyles.button, gameStyles.spinButton]} 
          onPress={handleSpin}
          disabled={isSpinning}>
          <Text style={gameStyles.buttonText}>
            {isSpinning ? '🔄 جاري...' : '🎰 العب'}
          </Text>
        </TouchableOpacity>
      </View>
    </View>
  );
};

export default function App() {
  const [isLoggedIn, setIsLoggedIn] = useState(false);
  const [isLoading, setIsLoading] = useState(true);

  useEffect(() => {
    checkLogin();
  }, []);

  const checkLogin = async () => {
    const token = await AsyncStorage.getItem('access_token');
    setIsLoggedIn(!!token);
    setIsLoading(false);
  };

  if (isLoading) {
    return (
      <View style={{ flex: 1, justifyContent: 'center', alignItems: 'center', backgroundColor: '#1a1a1a' }}>
        <Text style={{ color: '#FFD700', fontSize: 20 }}>جاري التحميل...</Text>
      </View>
    );
  }

  return (
    <NavigationContainer>
      <Stack.Navigator screenOptions={{ headerShown: false }}>
        {isLoggedIn ? (
          <Stack.Screen name="Main" component={MainDrawer} />
        ) : (
          <Stack.Screen name="Login" component={LoginScreen} />
        )}
      </Stack.Navigator>
    </NavigationContainer>
  );
}
